## 绕过 UniversalFragmentPBR

到目前为止，我们的 `Fragment` 函数一直依赖 URP 的 `UniversalFragmentPBR` 来做所有光照计算。它很方便，但也是一个黑盒。

要编写自定义光照，我们需要手动获取光源数据并自行计算颜色。让我们从重写 `Fragment` 函数开始。

### 当前结构

```hlsl
half4 Fragment(Interpolators input) : SV_TARGET
{
    // 1. 准备 InputData
    InputData inputData = (InputData)0;
    inputData.positionWS = input.positionWS;
    inputData.normalWS = normalize(input.normalWS);
    inputData.viewDirectionWS = GetWorldSpaceNormalizeViewDir(input.positionWS);
    inputData.shadowCoord = TransformWorldToShadowCoord(input.positionWS);
    
    // 2. 准备 SurfaceData
    SurfaceData surfaceData = (SurfaceData)0;
    // ... 设置 albedo, metallic, smoothness 等 ...
    
    // 3. 调用 URP 内置 PBR
    return UniversalFragmentPBR(inputData, surfaceData);
}
```

### 自定义结构

让我们用自定义光照替换第 3 步。首先，创建一个结构体来保存我们的自定义表面数据：

```hlsl
struct MySurfaceData
{
    half3 albedo;
    half3 emission;
    half  metallic;
    half  smoothness;
    half  occlusion;
    half  alpha;
    half3 normalWS;
    half3 viewDirectionWS;
    float3 positionWS;
};
```

然后，创建一个光照数据结构体：

```hlsl
struct MyLightingData
{
    half3 normalWS;
    half3 viewDirectionWS;
    half3 reflectionDirectionWS;
    half  NdotV;
    half  fresnel;
};
```

## 构建自定义光照函数

### 获取光源数据

我们需要一个函数来初始化光照数据：

```hlsl
MyLightingData CreateLightingData(MySurfaceData surface, InputData inputData)
{
    MyLightingData data = (MyLightingData)0;
    
    data.normalWS = surface.normalWS;
    data.viewDirectionWS = surface.viewDirectionWS;
    data.reflectionDirectionWS = reflect(-surface.viewDirectionWS, surface.normalWS);
    data.NdotV = saturate(dot(surface.normalWS, surface.viewDirectionWS));
    data.fresnel = pow(1 - data.NdotV, 5);
    
    return data;
}
```

### 漫反射模型

漫反射是光照模型的核心。让我们从简单的 Lambert 开始，然后扩展到更高级的模型：

```hlsl
// Lambert 漫反射（基础）
half3 LambertDiffuse(half3 lightColor, half3 lightDir, half3 normalWS)
{
    half NdotL = saturate(dot(normalWS, lightDir));
    return lightColor * NdotL;
}

// Half-Lambert（更好地利用阴影区域）
half3 HalfLambertDiffuse(half3 lightColor, half3 lightDir, half3 normalWS)
{
    half NdotL = dot(normalWS, lightDir) * 0.5 + 0.5;
    return lightColor * NdotL;
}

// 带可调能量的 Half-Lambert
half3 EnergyHalfLambert(half3 lightColor, half3 lightDir, half3 normalWS, half energy)
{
    half NdotL = dot(normalWS, lightDir) * 0.5 + 0.5;
    NdotL = pow(NdotL, energy);  // energy > 1 = 更对比
    return lightColor * NdotL;
}
```

### 高光模型

对于高光，我们可以实现多种模型：

```hlsl
// Blinn-Phong 高光
half3 BlinnPhongSpecular(half3 lightColor, half3 lightDir, half3 normalWS, half3 viewDir, half smoothness)
{
    half3 halfVector = normalize(lightDir + viewDir);
    half NdotH = saturate(dot(normalWS, halfVector));
    half shininess = exp2(smoothness * 10 + 1);
    half spec = pow(NdotH, shininess);
    return lightColor * spec;
}

// GGX/Trowbridge-Reitz（PBR 标准）
half3 GGXSpecular(half3 lightColor, half3 lightDir, half3 normalWS, half3 viewDir, half roughness)
{
    half3 halfVector = normalize(lightDir + viewDir);
    half NdotH = saturate(dot(normalWS, halfVector));
    half NdotV = saturate(dot(normalWS, viewDir));
    half NdotL = saturate(dot(normalWS, lightDir));
    
    half a2 = roughness * roughness;
    half d = NdotH * NdotH * (a2 - 1) + 1;
    half D = a2 / (3.14159 * d * d);
    
    // 简化的 Fresnel (Schlick)
    half3 F = lerp(half3(0.04, 0.04, 0.04), half3(1, 1, 1), pow(1 - NdotV, 5));
    
    return lightColor * D * F * NdotL;
}
```

## 卡通渲染（Toon Shading）

让我们实现第一个完整的自定义光照模型：**卡通渲染（Toon Shading）**，也叫 Cel Shading。

### Cel Shading

卡通渲染的核心思想是用**阶梯函数**替代平滑的漫反射过渡，产生平坦的色带。

```hlsl
// 简单的二值卡通漫反射
half3 ToonDiffuse(half3 lightColor, half3 lightDir, half3 normalWS, half threshold)
{
    half NdotL = dot(normalWS, lightDir);
    half toon = step(threshold, NdotL);  // 硬边阶梯
    return lightColor * toon;
}
```

但硬边通常太突兀了。让我们用一个平滑的阶梯函数来改进：

```hlsl
// 平滑卡通漫反射
half3 SmoothToonDiffuse(half3 lightColor, half3 lightDir, half3 normalWS, half threshold, half softness)
{
    half NdotL = dot(normalWS, lightDir) * 0.5 + 0.5;  // 映射到 [0,1]
    half toon = smoothstep(threshold - softness, threshold + softness, NdotL);
    return lightColor * toon;
}
```

### Ramp 贴图

更灵活的方法是使用**渐变贴图（Ramp Texture）**——一张一维纹理，定义了漫反射响应曲线。

```hlsl
// 在属性中添加
// _ToonRamp("Toon Ramp", 2D) = "white" {}

// 在 HLSL 中
TEXTURE2D(_ToonRamp);
SAMPLER(sampler_ToonRamp);

half3 RampDiffuse(half3 lightColor, half3 lightDir, half3 normalWS)
{
    half NdotL = dot(normalWS, lightDir) * 0.5 + 0.5;
    half3 rampColor = SAMPLE_TEXTURE2D(_ToonRamp, sampler_ToonRamp, float2(NdotL, 0.5)).rgb;
    return lightColor * rampColor;
}
```

> **技巧：** Ramp 贴图让美术完全控制光照响应。想要硬边卡通？画一个阶梯状渐变。想要柔和风格化？画一个平滑曲线。想要双色效果？画两个色带。

### 完整卡通着色器片段

```hlsl
half4 Fragment(Interpolators input) : SV_TARGET
{
    // 准备数据
    InputData inputData = (InputData)0;
    inputData.positionWS = input.positionWS;
    inputData.normalWS = normalize(input.normalWS);
    inputData.viewDirectionWS = GetWorldSpaceNormalizeViewDir(input.positionWS);
    inputData.shadowCoord = TransformWorldToShadowCoord(input.positionWS);
    
    // 采样纹理
    half4 colorSample = SAMPLE_TEXTURE2D(_ColorMap, sampler_ColorMap, input.uv);
    half3 albedo = colorSample.rgb * _ColorTint.rgb;
    
    // 法线贴图
    #if defined(_NORMALMAP)
        half3 normalTS = UnpackNormalScale(
            SAMPLE_TEXTURE2D(_BumpMap, sampler_BumpMap, input.uv), _NormalScale);
        // ... 转换到世界空间 ...
    #else
        half3 normalWS = inputData.normalWS;
    #endif
    
    // 主光源
    Light mainLight = GetMainLight(inputData.shadowCoord, inputData.positionWS, half4(1,1,1,1));
    
    // 卡通漫反射
    half NdotL = dot(normalWS, mainLight.direction) * 0.5 + 0.5;
    half toon = smoothstep(_CelShadeMidPoint - _CelShadeSoftness, 
                           _CelShadeMidPoint + _CelShadeSoftness, NdotL);
    
    // 应用阴影衰减
    toon *= lerp(1, mainLight.shadowAttenuation, _ReceiveShadowMappingAmount);
    
    half3 diffuse = mainLight.color * toon * albedo;
    
    // 卡通高光（锐利）
    half3 halfVector = normalize(mainLight.direction + inputData.viewDirectionWS);
    half NdotH = saturate(dot(normalWS, halfVector));
    half specToon = step(_SpecularThreshold, NdotH);
    half3 specular = mainLight.color * specToon * _SpecularColor.rgb;
    
    // 自发光
    half3 emission = SAMPLE_TEXTURE2D(_EmissionMap, sampler_EmissionMap, input.uv).rgb 
                  * _EmissionTint.rgb;
    
    // 环境光（简化）
    half3 ambient = SampleSH(normalWS) * albedo * _AmbientStrength;
    
    half3 finalColor = diffuse + specular + emission + ambient;
    
    return half4(finalColor, colorSample.a * _ColorTint.a);
}
```

### 卡通属性

```hlsl
// 在 MyLit.shader 中

_CelShadeSoftness("Cel Softness", Range(0, 0.5)) = 0.05
```

## 布料光照（Cloth）

布料（如丝绸、棉布、天鹅绒）有独特的光照特性。它们通常具有**各向异性**高光——高光沿织物纤维方向拉伸。

```hlsl
half3 ClothDiffuse(half3 lightColor, half3 lightDir, half3 normalWS, half wrapAmount)
{
    // "Wrap" 漫反射 —— 让光在边缘弯曲，模拟次表面散射
    half NdotL = dot(normalWS, lightDir);
    half wrapped = (NdotL + wrapAmount) / (1 + wrapAmount);
    return lightColor * saturate(wrapped);
}

half3 ClothSpecular(half3 lightColor, half3 lightDir, half3 normalWS, half3 viewDir, 
                    half3 tangentWS, half roughness, half anisotropy)
{
    // 各向异性高光（Ward 模型简化版）
    half3 halfVector = normalize(lightDir + viewDir);
    
    half NdotH = dot(normalWS, halfVector);
    half TdotH = dot(tangentWS, halfVector);
    half NdotV = saturate(dot(normalWS, viewDir));
    half NdotL = saturate(dot(normalWS, lightDir));
    
    // 沿切线方向拉伸高光
    half aniso = sqrt(max(1 - TdotH * TdotH, 0));
    half exponent = roughness * 100;
    half spec = pow(aniso, exponent) * anisotropy;
    
    return lightColor * spec * NdotL;
}
```

> **用途：** 这种模型非常适合丝绸、缎子、棉布，以及任何有方向性微观几何结构的表面。

## 皮肤光照（Skin Subsurface）

皮肤不是完全不透明的——光会穿透表面，在内部散射，然后从附近的点出来。这就是**次表面散射（Subsurface Scattering, SSS）**。

```hlsl
half3 SkinDiffuse(half3 lightColor, half3 lightDir, half3 normalWS, half3 viewDir)
{
    // 标准漫反射
    half NdotL = dot(normalWS, lightDir);
    
    // 包裹光照（Wrap Lighting）
    half wrap = (NdotL + 0.5) / 1.5;  // 让光"弯曲"到背面
    
    // 次表面散射近似
    half3 scatterColor = half3(0.8, 0.3, 0.2);  // 暖色散射
    half scatter = pow(saturate(NdotL * 0.5 + 0.5), 2) * 0.3;
    
    half3 diffuse = lightColor * (saturate(wrap) + scatter * scatterColor);
    return diffuse;
}

half3 SkinSpecular(half3 lightColor, half3 lightDir, half3 normalWS, half3 viewDir)
{
    // 皮肤的镜面高光很宽很柔和（不是锐利的 GGX）
    half3 halfVector = normalize(lightDir + viewDir);
    half NdotH = saturate(dot(normalWS, halfVector));
    
    // 使用低指数 = 宽高光
    half spec = pow(NdotH, 16) * 0.3;  // 宽而弱
    
    // 仅使用红色的 Fresnel（皮肤特性）
    half fresnel = pow(1 - saturate(dot(normalWS, viewDir)), 3);
    half3 skinFresnel = half3(0.8, 0.4, 0.3) * fresnel;
    
    return lightColor * (spec + skinFresnel * 0.1);
}
```

> **提示：** 对于更高质量的 SSS，URP 提供了 `SampleTransmission` 函数和 `SurfaceData` 中的 `transmission` 字段。这里展示的是简化的近似方法。

## 头发光照（Anisotropic Hair）

头发有独特的高光模式——高光沿头发方向拉伸成两个条带（R lobe 和 TT lobe）。

```hlsl
half3 AnisotropicHairSpecular(half3 lightColor, half3 lightDir, half3 normalWS, 
                              half3 viewDir, half3 tangentWS, half roughness)
{
    half3 halfVector = normalize(lightDir + viewDir);
    
    // 沿切线方向（头发方向）
    half TdotH = dot(tangentWS, halfVector);
    half NdotH = dot(normalWS, halfVector);
    
    // R 波瓣（反射波瓣）—— 沿切线方向
    half rLobe = sqrt(1 - TdotH * TdotH);
    half rExponent = lerp(2, 64, 1 - roughness);
    half rSpec = pow(rLobe, rExponent);
    
    // TT 波瓣（透射波瓣）—— 在背面
    half TTdotH = dot(-tangentWS, halfVector);
    half ttLobe = sqrt(1 - TTdotH * TTdotH);
    half ttSpec = pow(ttLobe, 32) * 0.5;
    
    half3 specular = lightColor * (rSpec + ttSpec);
    return specular;
}
```

> **Kajiya-Kay 模型：** 这是头发光照的经典模型。更现代的模型（如 Marschner）更复杂，但 Kajiya-Kay 在性能和质量之间取得了很好的平衡。

## 植物叶片（Foliage）

植物叶片是半透明的——背光时会"发光"。它们还需要双面渲染和简化的高光。

```hlsl
half3 FoliageDiffuse(half3 lightColor, half3 lightDir, half3 normalWS, 
                     half3 backNormalWS, half3 viewDir)
{
    // 正面光照
    half NdotL = dot(normalWS, lightDir);
    
    // 半 Lambert（正面）
    half front = NdotL * 0.5 + 0.5;
    
    // 背面透射（光穿透叶片）
    half back = saturate(dot(backNormalWS, lightDir) * 0.5 + 0.5);
    half transmission = pow(back, 2) * 0.5;  // 光穿透叶片
    
    // 叶片特有的暖色透射
    half3 transmissionColor = half3(0.6, 0.8, 0.2);  // 绿黄色
    
    return lightColor * (front + transmission * transmissionColor);
}

half3 FoliageSpecular(half3 lightColor, half3 lightDir, half3 normalWS, half3 viewDir)
{
    // 叶片高光：非常宽、非常弱、偏蓝绿色
    half3 halfVector = normalize(lightDir + viewDir);
    half NdotH = saturate(dot(normalWS, halfVector));
    half spec = pow(NdotH, 4) * 0.15;  // 宽而暗淡
    
    half3 specColor = half3(0.7, 0.9, 0.6);  // 蓝绿色
    return lightColor * spec * specColor;
}
```

> **提示：** 对于植物，记得在检视器中设置 `Cull Off`（双面渲染），并考虑使用 `_DOUBLE_SIDED_NORMALS` 关键字来翻转背面法线。

## 调试可视化

当你编写自定义光照时，调试变得至关重要。让我们添加一些调试视图。

### 可视化函数

```hlsl
half4 DebugView(half3 value, int mode)
{
    switch (mode)
    {
        case 0: return half4(value, 1);           // 直接输出
        case 1: return half4(value.xxx, 1);       // 仅红色通道
        case 2: return half4(value.yyy, 1);       // 仅绿色通道
        case 3: return half4(value.zzz, 1);       // 仅蓝色通道
        case 4: return half4(dot(value, half3(0.3, 0.59, 0.11)).xxx, 1); // 亮度
        case 5: return half4(NormalizeToColor(value), 1); // 向量转颜色
        default: return half4(value, 1);
    }
}

half3 NormalizeToColor(half3 v)
{
    return v * 0.5 + 0.5;  // [-1,1] -> [0,1]
}
```

```hlsl

#endif
```

## 混合光照模型

最强大的技术是将多个模型混合。例如，一个角色着色器可能混合 PBR（金属部件）、卡通（衣服）和 SSS（皮肤）：

```hlsl
half3 MixedLighting(MySurfaceData surface, MyLightingData lighting, Light light)
{
    half NdotL = dot(lighting.normalWS, light.direction);
    
    // PBR 分量
    half3 pbr = GGXSpecular(light.color, light.direction, lighting.normalWS, 
                              lighting.viewDirectionWS, 1 - surface.smoothness);
    
    // 卡通分量
    half toon = smoothstep(0.5 - 0.05, 0.5 + 0.05, NdotL * 0.5 + 0.5);
    
    // SSS 分量
    half wrap = (NdotL + 0.5) / 1.5;
    half3 sss = light.color * wrap * half3(0.8, 0.3, 0.2) * 0.3;
    
    // 根据金属度混合
    half3 diffuse = lerp(toon * light.color, pbr, surface.metallic);
    diffuse += sss * (1 - surface.metallic);
    
    return diffuse * surface.albedo;
}
```

## 总结

### 光照模型速查表

| 模型 | 核心公式 | 适用材质 |
|------|---------|----------|
| **Lambert** | `NdotL` | 哑光表面、墙壁 |
| **Half-Lambert** | `NdotL * 0.5 + 0.5` | 游戏通用、卡通 |
| **Cel/Toon** | `smoothstep(threshold, NdotL)` | 卡通、风格化 |
| **Ramp** | `tex2D(ramp, NdotL)` | 美术驱动的风格化 |
| **GGX/PBR** | `D * F * G / (4*NdotL*NdotV)` | 写实金属、塑料 |
| **Ward (各向异性)** | `pow(aniso, exp)` | 布料、拉丝金属 |
| **Kajiya-Kay** | `sqrt(1-TdotH²)` | 头发、毛皮 |
| **Wrap/SSS** | `(NdotL + w) / (1+w)` | 皮肤、蜡、玉石 |
| **Foliage** | `front + back² * tint` | 叶片、植物 |

```
├── 9. 应用自发光 + 遮挡
├── 10. 调试可视化（可选）
└── 11. 返回最终颜色
```

### 性能提示

- **避免分支：** GPU 不擅长 `if` 语句。使用 `lerp` 和 `step` 替代
- **预计算：** 在顶点阶段计算可以计算的东西
- **纹理查找昂贵：** 每个纹理采样都有成本。打包遮罩纹理以减少采样次数
- **关键字控制：** 用 `shader_feature` 让美术可以禁用不需要的光照特性
